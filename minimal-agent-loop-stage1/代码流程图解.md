```
}
```

```text
FUNCTION main() -> 退出码:
    args   = 解析命令行
    configure_logging(args.quiet)

    task = args.task
    IF task 为空 THEN
        task = input("请输入要交给 Agent 的任务：").strip()

    TRY:
        settings = Settings.from_env()        # ← 可能抛 ValueError（配置缺失）
        agent    = MinimalAgent(settings)     # ← 可能抛 RuntimeError（缺 openai 包）
        result   = agent.run(task)
    CATCH (ValueError, RuntimeError) AS error:   # 只接这两类，真 bug 让它崩
        result = {status:"error", answer:str(error), tools_used:[], steps_summary:[]}
        #        ↑ 故意造成和 agent 成功时完全同构的形状

    PRINT "=== Agent 最终结构化结果 ==="
    PRINT json(result, 中文不转义, 缩进2)

    IF result.status == "success" THEN RETURN 0
    ELSE                              RETURN 1     # 可直接用退出码判断成败

IF 作为脚本运行 THEN 退出(main())
```

**记忆点**：入口完全不关心 agent 内部发生了什么，它只依赖第 0 节那个四字段契约。

```text
CONST SYSTEM_INSTRUCTIONS = "
    1. 按需用工具，不要为了调用而调用

# ---------- agent.py:40-57 最终答案的强约束格式 ----------
CONST FINAL_TEXT_FORMAT = {
    format: {
        type: "json_schema", name: "agent_final_answer", strict: true,
        schema: 对象{
            status:        枚举["success","error"],
            answer:        字符串,
            tools_used:    字符串数组,
            steps_summary: 字符串数组(最多12条),
            必填: 以上全部, 禁止额外字段
        }
    }
}

# ---------- agent.py:60-68 统一错误结果 ----------
FUNCTION _error_result(message, tools_used):
    返回 {status:"error", answer:message, tools_used:tools_used, steps_summary:[]}

# ---------- agent.py:71-101 本地防御性校验 ----------
FUNCTION _parse_final_json(text, tools_used):
    TRY: result = json.loads(text)
    CATCH 解析失败: 返回 _error_result("不是有效 JSON：" + 错误, tools_used)

    # API 已经用 strict schema 保证过一次，这里再手工验一次
    # 原因：中转/代理服务不一定真支持 Structured Outputs，SDK 行为也会变
    IF 不是字典 OR 字段集合 != {status, answer, tools_used, steps_summary}:
        返回 _error_result("字段不符合约定", tools_used)
    IF result.status 不在 {"success","error"}:   返回 _error_result(...)
    IF answer 不是字符串:                        返回 _error_result(...)
    IF tools_used 不是「全字符串的列表」:         返回 _error_result(...)
    IF steps_summary 不是「全字符串的列表」:      返回 _error_result(...)

    result.tools_used = tools_used      # ★ 用程序账本覆盖模型自报，防漏报/虚构
    返回 result
```

### 2.3 `run()` 主循环

```text
# ---------- agent.py:125-206 ----------
FUNCTION MinimalAgent.run(user_task) -> 结果字典:

    # ---------- 准备 ----------
    IF user_task 不是非空字符串 THEN 返回 _error_result("用户任务不能为空", [])
    started_at = monotonic()            # 全程唯一时间基准（不受系统改时间影响）
    tools_used = []                     # 程序自己的工具账本
    input_items = [ {role:"user", content: user_task.strip()} ]
    #             ↑ 不是普通消息列表，而是 Responses API 的 input 数组，
    #               后面会被塞进「模型原始输出对象」和「工具输出对象」

    # ---------- 循环：步数闸门 ----------
    FOR step IN 1 .. settings.max_steps:

        # ---- 动作1：总超时闸门（唯一的时间总预算）----
        remaining = settings.total_timeout_seconds - (monotonic() - started_at)
        IF remaining <= 0 THEN 返回 _error_result("达到总超时限制", tools_used)

        LOG "[Step step/max_steps] 请求模型，剩余总时间 remaining 秒"

        # ---- 动作2+3：调模型（单次超时被总预算钳制）----
        TRY:
            request_timeout = max(1.0, min(settings.api_timeout_seconds, remaining))
            #                 ↑下限1秒防抖动   ↑上限：单次永远不能超过剩余总时间
            response = self.client
                           .with_options(timeout=request_timeout)
                           .responses.create(
                               model        = settings.model,
                               instructions = SYSTEM_INSTRUCTIONS,   # 等价 system prompt
                               input        = input_items,           # 累积的轨迹
                               tools        = TOOL_SCHEMAS,          # 只传 schema，不传函数
                               tool_choice  = "auto",
                               text         = FINAL_TEXT_FORMAT)     # 约束最终输出
        CATCH 任何异常 AS error:
            LOG 堆栈
            返回 _error_result("LLM API 调用失败：" + 类型 + 错误, tools_used)
            # ↑ 注意：API 出错不重试循环，直接结束（重试交给 SDK 的 max_retries）

        # ---- 动作4：回灌轨迹 + 分流 ----
        input_items.extend(response.output)      # ★ 必须先放模型输出
        function_calls = [item FOR item IN response.output
                                 IF item.type == "function_call"]

        # ---- 动作5a：没有工具调用 → 模型认为完成了 ----
        IF function_calls 为空:
            IF response.output_text 为空 THEN
                返回 _error_result("既没调工具，也没返回最终答案", tools_used)
            LOG "Agent 在第 step 步完成任务"
            返回 _parse_final_json(response.output_text, tools_used)
            # ↑ 这里 return，循环结束

        # ---- 动作5b：有工具调用 → 逐个执行（一个响应可能含多个）----
        FOR tool_call IN function_calls:
            tool_name = tool_call.name
            LOG "调用工具：tool_name，参数：tool_call.arguments"

            TRY: arguments = json.loads(tool_call.arguments)   # 模型给的是 JSON 字符串
            CATCH 解析失败:
                # ★ 不抛出、不中断，而是把错误本身当成工具结果回传，
                #   让模型下一步自己看见并修正参数
                tool_result = {ok:false, tool:tool_name,
                               error:"工具参数不是有效 JSON：" + 错误}
            ELSE:
                # ---- 工具执行前再查一次总超时（时间检查粒度够细）----
                remaining = settings.total_timeout_seconds - (monotonic() - started_at)
                IF remaining <= 0 THEN
                    返回 _error_result("在工具执行前达到总超时限制", tools_used)

                tool_timeout = min(settings.tool_timeout_seconds, remaining)
                #              ↑单工具超时同样不能超过剩余总时间
                tool_result = execute_tool(tool_name, arguments, timeout=tool_timeout)

            tools_used.append(tool_name)        # ★ 每个 function_call 记一次（会重复）
            LOG "工具结果：" + serialize_tool_result(tool_result)

            input_items.append({                # ★ 再 append 工具输出，靠 call_id 配对
                type:    "function_call_output",
                call_id: tool_call.call_id,     # ← 必须和上面那个 function_call 对上
                output:  serialize_tool_result(tool_result),   # 序列化成字符串
            })

    # ---------- 循环自然走完 = 步数用尽 ----------
    返回 _error_result("达到最大步数 max_steps，为防止无限循环已停止", tools_used)
```

### 2.4 三种超时在循环中的位置

```text
run() 开始 ──┬─ started_at = monotonic()
             │
    ┌────────▼───────────────────────────────────────────┐
    │ step 循环                                          │
    │   ①  remaining = total - 已用     ← 总超时闸门     │
    │   ②  API 超时 = max(1, min(api, remaining))        │
    │   ③  responses.create(...)                         │
    │   ④  无工具调用? ──是──▶ 解析 JSON 并 return       │
    │   ⑤  对每个工具:                                   │
    │         remaining = total - 已用  ← 总超时闸门     │
    │         工具超时 = min(tool, remaining)            │
    │         execute_tool(..., timeout=工具超时)         │
    └────────────────────────────────────────────────────┘
             │
    max_steps 次后仍无最终答案 ──▶ 返回「达到最大步数」
```

### 3.2 `calculator`：用 AST 白名单代替 eval

```text
# ---------- tools.py:58-70 白名单表 ----------
CONST 二元运算白名单 = {Add:加法, Sub:减法, Mult:乘法, Div:真除法,
                        FloorDiv:整除, Mod:取模, Pow:幂}
CONST 一元运算白名单 = {UAdd:取正, USub:取负}

# ---------- tools.py:73-95 递归求值（核心安全点）----------
FUNCTION _evaluate_math_node(node):
    IF node 是 Expression      THEN 返回 _evaluate_math_node(node.body)

    IF node 是 Constant 且值是 int/float
                               THEN 返回 node.value

    IF node 是 UnaryOp 且 op 在白名单
                               THEN 返回 白名单[node.op]( _evaluate_math_node(node.operand) )

    IF node 是 BinOp 且 op 在白名单:
        left  = _evaluate_math_node(node.left)
        right = _evaluate_math_node(node.right)
        IF node.op 是 Pow 且 |right| > 10 THEN 抛 ToolInputError("指数绝对值不能超过 10")
        result = 白名单[node.op](left, right)
        IF |result| > 1e100 THEN 抛 ToolInputError("计算结果过大")
        返回 result

    # 其他一切节点（Call / Name / Attribute / Subscript / Lambda…）直接拒绝
    抛 ToolInputError("表达式包含不允许的语法，只支持数字和基本四则运算")

# ---------- tools.py:98-112 对外工具 ----------
FUNCTION calculator(expression) -> 字典:
    IF expression 不是非空字符串 THEN 抛 ToolInputError
    IF 长度 > 200                THEN 抛 ToolInputError

    TRY:
        tree   = ast.parse(expression, mode:"eval")   # 只解析，绝不 eval/exec
        result = _evaluate_math_node(tree)
    CATCH (语法错误, 除零, 溢出) AS 错误:
        抛 ToolInputError("无法计算该表达式：" + 错误)

    返回 {expression: expression, result: result}
```

### 3.3 `read_file`：沙箱边界

```text
# ---------- tools.py:115-126 路径解析（所有文件类工具共用）----------
FUNCTION _resolve_sandbox_path(relative_path) -> 绝对路径:
    IF relative_path 不是非空字符串 THEN 抛 ToolInputError

    candidate = (SANDBOX_ROOT / relative_path).resolve()
    #           ↑ resolve 会展开 .. 和符号链接，所以能同时防住
    #             "…/../../etc/passwd" 和「软链指向外部」两种绕过

    TRY: candidate.relative_to(SANDBOX_ROOT)     # 不是真的子路径就抛 ValueError
    CATCH ValueError: 抛 ToolPermissionError("禁止读取 sandbox_files 之外的文件")
    返回 candidate
    # ★ 比字符串 startswith 检查可靠得多

# ---------- tools.py:129-149 ----------
FUNCTION read_file(path) -> 字典:
    file_path = _resolve_sandbox_path(path)

    IF 不是文件           THEN 抛 ToolInputError("文件不存在：" + path)
    IF 后缀不在白名单      THEN 抛 ToolPermissionError("不允许读取该文件类型")
    IF 大小 > 50KB        THEN 抛 ToolPermissionError("文件超过限制")

    TRY: content = 以 UTF-8 读取
    CATCH 解码失败: 抛 ToolInputError("文件不是 UTF-8 文本")

    返回 {path: 相对沙箱的规范化路径,
          content: content,
          characters: 字符数}
```

### 3.4 `search`：纯本地关键词搜索

```text
# ---------- tools.py:152-186 ----------
FUNCTION search(query, max_results) -> 字典:
    IF query 不是非空字符串              THEN 抛 ToolInputError
    IF max_results 不是 1..10 的整数     THEN 抛 ToolInputError

    keywords = [w.lower() FOR w IN query按空格切分 IF w非空]
    results  = []

    FOR file_path IN sorted(SANDBOX_ROOT.rglob("*")):     # 递归遍历沙箱
        IF 不是文件 OR 后缀不在白名单 OR 大小 > 50KB THEN 跳过
        TRY: lines = 读文本并按行切分
        CATCH 解码失败: 跳过

        FOR (行号, 行) IN 枚举(lines, 从1开始):
            IF 所有 keyword 都在 行小写 中:      # ★ 多关键词是 AND 关系
                results.append({path: 相对路径, line: 行号, snippet: 行[:300]})
                IF results 数量 >= max_results THEN
                    返回 {query: query, results: results}    # 提前返回
    返回 {query: query, results: results}
```

### 3.6 执行 + 权限 + 超时

```text
# ---------- tools.py:260-268 白名单闸门 ----------
FUNCTION _call_tool(tool_name, arguments):
    tool = TOOL_REGISTRY.get(tool_name)
    IF tool 是 None THEN
        抛 ToolNotFoundError("工具未注册或未授权：" + tool_name)
        # ★ 模型幻觉出 delete_file 也只会拿到这条错误，而不会被执行

    IF arguments 不是字典 THEN 抛 ToolInputError("参数必须是 JSON 对象")

    返回 tool.function(**arguments)      # Python 侧唯一真正调用工具的地方

# ---------- tools.py:271-295 统一出口：永远返回同形状字典 ----------
FUNCTION execute_tool(tool_name, arguments, timeout) -> 字典:
    executor = 线程池(最大1个线程)
    future   = executor.submit(_call_tool, tool_name, arguments)

    TRY:
        value = future.result(timeout = timeout)
        返回 {ok:true,  tool:tool_name, data: value}

    CATCH 超时:
        future.cancel()
        返回 {ok:false, tool:tool_name, error:"工具执行超过 timeout 秒"}
        # ⚠ 线程无法被强杀：cancel() 对已在跑的任务返回 False，
        #   线程会继续跑完，只是结果被丢弃。本项目三个工具都有界所以无害；
        #   换成长耗时/高风险工具必须改为独立进程/容器/远程任务。

    CATCH ToolError  AS 错误: 返回 {ok:false, tool:tool_name, error: str(错误)}
    CATCH TypeError  AS 错误: 返回 {ok:false, tool:tool_name, error:"工具参数错误：" + 错误}
    CATCH 其他异常   AS 错误:                     # 最外层保护，单个工具不让整个进程崩
        LOG 堆栈
        返回 {ok:false, tool:tool_name, error:"工具内部错误：" + 类型名}

    FINALLY:
        executor.shutdown(wait:false, cancel_futures:true)   # 不阻塞等待

# ---------- tools.py:298-300 ----------
FUNCTION serialize_tool_result(result) -> 字符串:
    返回 json.dumps(result, 中文不转义)      # 变成字符串才能塞进 function_call_output
```

## 4. `config.py` —— 配置（90 行）

```text
# ---------- config.py:18 ----------
CONST PROJECT_ROOT = 本文件所在目录

# ---------- config.py:21-38 极简 .env 解析（刻意不依赖 python-dotenv）----------
FUNCTION load_dotenv(path):
    IF path 不存在 THEN 直接返回          # 没有 .env 也能跑（靠系统环境变量）

    FOR 每一行 IN 文件内容:
        line = 去掉首尾空白
        IF 空行 OR 以 "#" 开头 OR 不含 "=" THEN 跳过
        key, value = line.split("=", 最多切1次)    # ★ 只切第一个=，值里的=不被破坏
        value = value.strip().strip('"').strip("'")
        os.environ.setdefault(key, value)
        # ★★ 用 setdefault 而不是赋值：
        #    已存在的系统环境变量优先级更高，.env 不会覆盖它
        #    → 部署/容器注入的密钥永远赢

# ---------- config.py:41-52 校验辅助 ----------
FUNCTION _positive_int(name, default):
    value = int(os.getenv(name, default))     # 转不动就抛 ValueError
    IF value <= 0 THEN 抛 ValueError(name + " 必须大于 0")
    返回 value

FUNCTION _positive_float(name, default):      # 同上，只是转 float

# ---------- config.py:55-90 配置对象 ----------
@dataclass(frozen=True)                       # frozen：创建后谁都改不了
class Settings:
    api_key:                字符串
    model:                  字符串
    base_url:               字符串 或 None
    max_steps:              整数
    total_timeout_seconds:  浮点
    tool_timeout_seconds:   浮点
    api_timeout_seconds:    浮点

    @classmethod
    FUNCTION from_env() -> Settings:
        load_dotenv(PROJECT_ROOT / ".env")

        api_key  = 环境变量 OPENAI_API_KEY  去空白
        model    = 环境变量 OPENAI_MODEL    去空白
        base_url = 环境变量 OPENAI_BASE_URL 去空白 或 None    # 空串 → None，
        #                                                      agent 才好判断传不传

        IF api_key 为空 THEN 抛 ValueError("缺少 OPENAI_API_KEY，请先复制 .env.example 为 .env")
        IF model   为空 THEN 抛 ValueError("缺少 OPENAI_MODEL，请填写有权限的模型名")

        返回 Settings(
            api_key, model, base_url,
            max_steps              = _positive_int  ("AGENT_MAX_STEPS",    8),
            total_timeout_seconds  = _positive_float("AGENT_TOTAL_TIMEOUT", 120),
            tool_timeout_seconds   = _positive_float("AGENT_TOOL_TIMEOUT",  8),
            api_timeout_seconds    = _positive_float("AGENT_API_TIMEOUT",   45),
        )
```

### 4.1 四类参数与三道闸

```text
                 ┌─ max_steps = 8 ──────────────▶ 逻辑死循环闸（步数）
命令/环境变量 ────┼─ total_timeout = 120s ────────▶ 时间总预算闸
（系统env > .env） ├─ api_timeout   = 45s ─────────▶ 单次模型请求上限
                 └─ tool_timeout  = 8s ──────────▶ 单个工具执行上限

嵌套关系（agent.py 里被显式钳制）：
    total(120s) ─┬─ 每次 API  : max(1.0, min(api_timeout, remaining))
                 └─ 每次工具 : min(tool_timeout, remaining)
    → 任何单次操作都不可能突破剩余总时间
```

## 6. 全项目通用模式速查

| 模式 | 实现位置 | 为什么这么写 |
|---|---|---|
| 结果形状统一 | `_error_result` + `_parse_final_json` + `execute_tool` 各 catch 分支 | 调用方永远不用分支判断结构 |
| 双份工具定义绑一起 | `ToolSpec(function, schema)` | 防 schema 与实现漂移 |
| 白名单兜底 | `TOOL_REGISTRY.get()` | 防模型幻觉出未授权工具名 |
| 错误当信息回喂 | `agent.py:180-184` 参数 JSON 非法时造 `ok:false` | 模型能看见错误并自我修正 |
| 路径安全用 `relative_to` | `_resolve_sandbox_path` | 同时防 `../` 与符号链接绕过 |
| 不用 eval | `_evaluate_math_node` AST 递归白名单 | 杜绝任意代码执行 |
| 单次超时被总预算钳制 | `max(1.0, min(api, remaining))` / `min(tool, remaining)` | 任何一次操作都突破不了总时间 |
| 账本覆盖自报 | `result["tools_used"] = tools_used` | 不信任模型的自我汇报 |
| 本地复验 schema | `_parse_final_json` | 代理服务可能不支持 Structured Outputs |
| 延迟导入 | `agent.py:110-114` 函数内 import openai | 让工具测试无需安装 openai |

### 6.1 有意的简化（Stage 1 的边界）

1. **模型不知道自己的预算**：`input_items` 里从未注入「还剩几步/几秒」，所以模型无法自主规划，只能靠外层熔断。这是刻意的教学简化。
2. **线程超时杀不掉真正的长任务**：见 `tools.py:274-276` 注释；三个工具都有界，所以演示成立。
3. **`tools_used` 会重复**：按 `function_call` 计数（`agent.py:193`），同一工具调用两次就出现两次。
4. **步数用尽 = 全盘失败**：`max_steps` 走完时，之前所有工具结果被丢弃，返回 error。它是硬熔断器，不是「尽量多做」。
